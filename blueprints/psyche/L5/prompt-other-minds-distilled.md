# Other Minds & Social Cognition Distilled 레포지토리 제작 프롬프트

나는 "Other Minds & Social Cognition Distilled" 레포지토리를 만들려고 해.
"우리는 타인의 마음을 안다"고 말하는 것과, **타인의 마음에 대한 접근이 왜 원리적으로 간접적이고, 어떤 추론 과정이 어떻게 작동하는지** — 그리고 이 "타인의 마음 문제"가 인식론적 문제이면서 동시에 사회 인지의 알고리즘 문제이기도 한 이유를 — 이해하는 것은 다르다.
"거울 뉴런이 공감의 신경 기반이다"고 말하는 것과, **거울 뉴런 가설이 실제로 무엇을 주장하고, 어떤 증거가 있으며, 왜 대부분의 신경과학자가 공감의 충분한 설명으로 보지 않는지** — 그리고 공감의 실제 신경 기반이 어떻게 더 복잡한지를 — 추적하는 것은 다르다.
마음이론(ToM)을 "남의 마음 읽기"로 아는 것과, **ToM이 어떻게 발달하고, 자폐 스펙트럼에서 어떻게 달라지며, 어떤 실험이 ToM의 어떤 측면을 분리하는지** — 그리고 ToM이 왜 단일한 능력이 아니라 여러 구성 요소의 집합인지를 — 이해하는 것은 다르다.
타인의 마음을 아는 것이 어떻게 가능한지를 — 그리고 그 앎이 어떻게 항상 불완전하고, 그 불완전성이 공감·오해·편견·사랑과 어떻게 연결되는지를 — 끝까지 파헤치는 레포를 만들 거야.

- **실용적 해결** — 우리는 타인의 마음을 어떻게 실제로 다루는가, 추론의 알고리즘

### Chapter 2: 마음이론 (ToM) — 타인의 정신 상태 귀인 (6개 문서)

### Chapter 3: 거울 뉴런과 공감 — 신화의 해부 (5개 문서)

import numpy as np

# 1) ToM — 틀린 믿음 추론 알고리즘
class TheoryOfMind:
    def __init__(self, perspective_taking_depth=2):
        self.depth = perspective_taking_depth  # 2차 믿음까지
    
    def first_order_belief(self, agent, world_state):
        """
        1차 믿음: 에이전트가 세계 상태에 대해 무엇을 믿는가
        에이전트의 관찰 이력 기반
        """
        observed = agent.get('observations', [])
        belief = world_state if world_state in observed else None
        return belief
    
    def false_belief_attribution(self, agent, actual_state, agent_last_seen):
        """
        틀린 믿음 귀인: 에이전트는 마지막으로 본 상태를 믿는다
        Sally-Anne 과제의 알고리즘 버전
        """
        agent_belief = agent_last_seen  # 에이전트의 믿음 (실제와 다를 수 있음)
        is_false_belief = agent_belief != actual_state
        
        return {
            'agent_believes': agent_belief,
            'actual_state':   actual_state,
            'is_false_belief': is_false_belief
        }
    
    def second_order_belief(self, agent_A, agent_B, proposition):
        """
        2차 믿음: A는 B가 P를 믿는다고 생각한다
        → 재귀적 마음 읽기
        """
        B_belief = self.first_order_belief(agent_B, proposition)
        A_thinks_B_believes = B_belief  # 단순화
        return A_thinks_B_believes

# 2) 공감 정확도 모델
def empathy_accuracy(target_emotional_state, perceiver_prediction,
                      shared_experience_level=0.5):
    """
    공감 정확도: 지각자가 타겟의 감정을 얼마나 정확히 예측하는가
    shared_experience: 유사한 경험이 많을수록 높음 (공감 격차)
    """
    base_accuracy = 1 - np.abs(target_emotional_state - perceiver_prediction)
    
    # 공유 경험이 정확도에 기여
    adjusted_accuracy = base_accuracy * (0.5 + 0.5 * shared_experience_level)
    
    return adjusted_accuracy

# 공감 격차 시뮬레이션
states = np.linspace(0, 1, 10)
predictions_high_shared = states + np.random.randn(10) * 0.1  # 높은 공유 경험
predictions_low_shared  = states + np.random.randn(10) * 0.4  # 낮은 공유 경험

# 3) 의인화 경향 모델
def anthropomorphization_tendency(stimulus_properties, social_brain_sensitivity=0.7):
    """
    사회적 자극이 아니어도 마음을 귀인하는 경향
    Heider-Simmel 실험: 삼각형·원이 움직이면 의도를 귀인
    """
    motion_unpredictability = stimulus_properties.get('unpredictability', 0)
    contingent_motion       = stimulus_properties.get('contingency', 0)
    
    # 불규칙하고 반응적인 움직임 → 의인화
    anthropomorphism = social_brain_sensitivity * (
        0.4 * motion_unpredictability + 0.6 * contingent_motion)
    
    return anthropomorphism

# 4) 시뮬레이션 이론 vs 이론 이론
def social_reasoning(situation, self_experience, use_simulation=True):
    """
    시뮬레이션 이론: 자신을 상황에 투사
    이론 이론: 암묵적 규칙으로 추론
    """
    if use_simulation:
        # 자신의 경험에서 타인의 반응 예측
        predicted_response = self_experience.get(situation, None)
        if predicted_response is None:
            # 경험 없으면 이론으로 보완
            predicted_response = 0.5  # 중립 예측
    else:
        # 이론 기반: "상황 S에서 사람들은 보통 R을 한다"
        social_rules = {
            'threat': 'fear', 'reward': 'pleasure',
            'loss': 'sadness', 'injustice': 'anger'
        }
        predicted_response = social_rules.get(situation, 'neutral')
    
    return predicted_response

## 💡 핵심 분석 대상

```
거울 뉴런 실제 vs 신화:

  실제 발견 (Rizzolatti, 원숭이):
  F5 뉴런: 원숭이가 행동할 때 + 다른 원숭이/인간 행동을 볼 때 발화
  → 행동 관찰과 실행의 공유 표상

  신화 (대중화 과정에서):
  "거울 뉴런 = 공감의 신경 기반"
  "공감이 없으면 거울 뉴런이 없다"
  
  왜 신화인가:
  ├─ 원숭이 F5 뉴런 ≠ 인간 공감 회로 (직접 대응 불명확)
  ├─ fMRI "거울" 활성화 ≠ 단일 뉴런 거울 속성
  ├─ 자폐 = 거울 뉴런 결함 이론: 지지 증거 약함
  └─ 공감과 "거울" 활성화의 상관 낮음

ToM의 구성 요소:
  1. 틀린 믿음 귀인 (TPJ 우측): "그는 상자에 공이 있다고 생각한다"
  2. 의도 귀인 (mPFC): "그는 X를 하려고 했다"
  3. 감정 귀인 (섬엽·amygdala): "그는 슬프다"
  4. 메타인지 귀인 (vmPFC): "그는 자신이 X를 안다는 것을 안다"
  
  각각이 독립적으로 손상 가능:
  → 자폐: 틀린 믿음 귀인 어려움, 감정 귀인은 상대적으로 온전
  → 정신병질: 인지적 공감 있지만 정서적 공감 없음

타인의 마음 문제의 잔여:
  알고리즘이 설명하는 것:
  ├─ 어떻게 타인의 행동에서 믿음을 귀인하는가
  ├─ 어떻게 타인의 감정을 추론하는가
  └─ 어떻게 공유된 신경 표상이 공감을 가능하게 하는가
  
  설명되지 않는 것:
  → 타인에게 정말 경험이 있는가 (철학적 좀비 문제)
  → 나의 "공감" 경험이 타인의 것과 동일한 경험인가
  ⇒ 사회 인지가 아무리 발전해도 타인 의식의 확실성은 남는다
```
